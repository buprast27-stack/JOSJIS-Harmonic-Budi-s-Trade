    // Formatters
    function formatPrice(price) {
      if (!price || isNaN(price)) return '$0.00';
      if (price >= 1000) return '$' + price.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
      if (price >= 1) return '$' + price.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 4 });
      if (price >= 0.01) return '$' + price.toFixed(4);
      return '$' + price.toFixed(7);
    }

    function formatCompactUSD(val) {
      const abs = Math.abs(val);
      const sign = val < 0 ? '-' : '';
      if (abs >= 1e9) return `${sign}$${(abs / 1e9).toFixed(2)}B`;
      if (abs >= 1e6) return `${sign}$${(abs / 1e6).toFixed(2)}M`;
      if (abs >= 1e3) return `${sign}$${(abs / 1e3).toFixed(1)}K`;
      return `${sign}$${abs.toFixed(0)}`;
    }

    function formatSignedPct(val, decimals = 2) {
      const sign = val > 0 ? '+' : '';
      return `${sign}${val.toFixed(decimals)}%`;
    }

    function generateSyntheticKlines(coin, tf, count = 60) {
      const tfScale = { '5m': 0.0024, '15m': 0.0045, '1H': 0.011, '4H': 0.024, '24H': 0.048 }[tf] || 0.02;
      const seedBase = coin.symbol.split('').reduce((acc, ch) => acc + ch.charCodeAt(0), 0) + tf.charCodeAt(0) * 13 + tf.length * 7;

      function pseudoRand(i) {
        const x = Math.sin(seedBase * 19.31 + i * 7.13) * 10000;
        return x - Math.floor(x);
      }

      const candles = [];
      let curClose = coin.price;
      const trendBias = (coin.change24h / 100) * (tf === '24H' ? 0.55 : tf === '4H' ? 0.35 : 0.18);

      for (let i = 0; i < count; i++) {
        const r1 = pseudoRand(i);
        const r2 = pseudoRand(i + 101);
        const wave = Math.sin((i / 7) + (seedBase % 5)) * tfScale * 0.85;
        const stepChange = ((r1 - 0.5) * tfScale * 1.6) + (trendBias / count) + (wave * 0.25);

        const close = curClose;
        const open = close / (1 + stepChange);
        const wickUp = Math.abs(r2 * tfScale * 0.65) * Math.max(open, close);
        const wickDn = Math.abs((1 - r2) * tfScale * 0.65) * Math.min(open, close);
        const high = Math.max(open, close) + wickUp;
        const low = Math.min(open, close) - wickDn;

        const baseVol = (coin.vol24h * (TF_MULTIPLIERS[tf]?.vol || 1)) / 24;
        const volSpikeFactor = i === 0 ? (0.85 + pseudoRand(999) * 1.55) : (0.65 + r1 * 0.8);

        candles.unshift({ open, high, low, close, volume: baseVol * volSpikeFactor });
        curClose = open;
      }
      return candles;
    }

    // 1. EMA (Exponential Moving Average)
    function calculateEMA(closes, period) {
      if (!closes.length) return [];
      const k = 2 / (period + 1);
      const emaArr = [closes[0]];
      for (let i = 1; i < closes.length; i++) {
        emaArr.push(closes[i] * k + emaArr[i - 1] * (1 - k));
      }
      return emaArr;
    }

    // 2. Stochastic Oscillator (5, 3, 3)
    function calculateStochastic533(candles) {
      const kPeriod = 5, smoothK = 3, dPeriod = 3;
      if (candles.length < kPeriod + smoothK + dPeriod) {
        return { k: 50, d: 50 };
      }
      const rawK = [];
      for (let i = kPeriod - 1; i < candles.length; i++) {
        const slice = candles.slice(i - kPeriod + 1, i + 1);
        const highestHigh = Math.max(...slice.map(c => c.high));
        const lowestLow = Math.min(...slice.map(c => c.low));
        const denom = highestHigh - lowestLow;
        rawK.push(denom === 0 ? 50 : ((candles[i].close - lowestLow) / denom) * 100);
      }
      const slowK = [];
      for (let i = smoothK - 1; i < rawK.length; i++) {
        const sum = rawK.slice(i - smoothK + 1, i + 1).reduce((a, b) => a + b, 0);
        slowK.push(sum / smoothK);
      }
      const slowD = [];
      for (let i = dPeriod - 1; i < slowK.length; i++) {
        const sum = slowK.slice(i - dPeriod + 1, i + 1).reduce((a, b) => a + b, 0);
        slowD.push(sum / dPeriod);
      }
      return {
        k: slowK[slowK.length - 1] ?? 50,
        d: slowD[slowD.length - 1] ?? 50
      };
    }

    // 3. Smart Money Concept (SMC)
    function analyzeSmartMoneyConcept(candles) {
      const recent = candles.slice(-30);
      const highs = recent.map(c => c.high);
      const lows = recent.map(c => c.low);
      const swingHigh = Math.max(...highs);
      const swingLow = Math.min(...lows);
      const last = recent[recent.length - 1];
      const range = Math.max(1e-9, swingHigh - swingLow);
      const eq = swingLow + range * 0.5;
      const rangePosPct = ((last.close - swingLow) / range) * 100;

      const priorHigh = Math.max(...recent.slice(0, -3).map(c => c.high));
      const priorLow = Math.min(...recent.slice(0, -3).map(c => c.low));

      let structure = 'Range Consolidation';
      let bias = 0;
      if (last.close > priorHigh) {
        structure = 'Bullish BOS / CHoCH';
        bias += 1;
      } else if (last.close < priorLow) {
        structure = 'Bearish BOS / CHoCH';
        bias -= 1;
      }

      let bullishOB = swingLow + range * 0.14;
      let bearishOB = swingHigh - range * 0.14;
      for (let i = recent.length - 4; i >= 2; i--) {
        if (recent[i].close < recent[i].open && recent[i + 1].close > recent[i].high) {
          bullishOB = (recent[i].open + recent[i].low) / 2;
          break;
        }
      }
      for (let i = recent.length - 4; i >= 2; i--) {
        if (recent[i].close > recent[i].open && recent[i + 1].close < recent[i].low) {
          bearishOB = (recent[i].open + recent[i].high) / 2;
          break;
        }
      }

      const zoneLabel = rangePosPct <= 45
        ? `Discount (${rangePosPct.toFixed(0)}%)`
        : rangePosPct >= 55
          ? `Premium (${rangePosPct.toFixed(0)}%)`
          : `Equilibrium (${rangePosPct.toFixed(0)}%)`;

      if (rangePosPct <= 42) bias += 0.6;
      if (rangePosPct >= 58) bias -= 0.6;

      return { structure, zoneLabel, rangePosPct, bullishOB, bearishOB, swingHigh, swingLow, eq, bias };
    }

    // 4. Fibonacci Retracement (Golden Pocket 0.618 - 0.65)
    function analyzeFibonacci(swingHigh, swingLow, currentPrice, isUptrend) {
      const diff = Math.max(1e-9, swingHigh - swingLow);
      const fib0 = isUptrend ? swingHigh : swingLow;
      const fib382 = isUptrend ? swingHigh - diff * 0.382 : swingLow + diff * 0.382;
      const fib500 = swingLow + diff * 0.5;
      const fib618 = isUptrend ? swingHigh - diff * 0.618 : swingLow + diff * 0.618;
      const fib650 = isUptrend ? swingHigh - diff * 0.650 : swingLow + diff * 0.650;
      const fib786 = isUptrend ? swingHigh - diff * 0.786 : swingLow + diff * 0.786;
      const extNeg27 = isUptrend ? swingHigh + diff * 0.272 : swingLow - diff * 0.272;

      const levels = [
        { name: '0.0 (Swing)', val: fib0 },
        { name: '0.382', val: fib382 },
        { name: '0.500 (Eq)', val: fib500 },
        { name: '0.618 (Golden)', val: fib618 },
        { name: '0.786 (Deep)', val: fib786 }
      ];

      let nearest = levels[0];
      let minDist = Infinity;
      levels.forEach(lvl => {
        const d = Math.abs(currentPrice - lvl.val);
        if (d < minDist) {
          minDist = d;
          nearest = lvl;
        }
      });

      const gpLow = Math.min(fib618, fib650);
      const gpHigh = Math.max(fib618, fib650);
      const inGoldenPocket = (currentPrice >= gpLow * 0.992 && currentPrice <= gpHigh * 1.008) || (Math.abs(currentPrice - fib618) / diff <= 0.08);

      return { nearestLabel: nearest.name, nearestPrice: nearest.val, inGoldenPocket, fib618, fib650, fib500, fib0, extNeg27, isUptrend };
    }

    // 5. Harmonic Pattern Detector (XABCD: Gartley, Bat, Butterfly, Crab, Cypher)
    function analyzeHarmonicPattern(candles) {
      const slice = candles.slice(-36);
      const segSize = Math.floor(slice.length / 4);
      const s1 = slice.slice(0, segSize);
      const s2 = slice.slice(segSize, segSize * 2);
      const s3 = slice.slice(segSize * 2, segSize * 3);
      const s4 = slice.slice(segSize * 3);

      const lastClose = slice[slice.length - 1].close;
      const midPrice = slice[Math.floor(slice.length / 2)].close;
      const isBullishSetup = lastClose <= midPrice;

      let X, A, B, D;
      if (isBullishSetup) {
        X = Math.min(...s1.map(c => c.low));
        A = Math.max(...s2.map(c => c.high));
        B = Math.min(...s3.map(c => c.low));
        D = lastClose;
      } else {
        X = Math.max(...s1.map(c => c.high));
        A = Math.min(...s2.map(c => c.low));
        B = Math.max(...s3.map(c => c.high));
        D = lastClose;
      }

      const xa = Math.max(1e-9, Math.abs(A - X));
      const abRetrace = Math.abs(B - A) / xa;
      const adRetrace = Math.abs(D - A) / xa;

      let shortCode = isBullishSetup ? 'Bull Cypher' : 'Bear Cypher';
      let patternName = `${isBullishSetup ? 'Bullish' : 'Bearish'} Cypher`;

      if (adRetrace >= 1.45) {
        shortCode = isBullishSetup ? 'Bull Crab' : 'Bear Crab';
        patternName = `${isBullishSetup ? 'Bullish' : 'Bearish'} Crab (1.618)`;
      } else if (adRetrace >= 1.15) {
        shortCode = isBullishSetup ? 'Bull Butterfly' : 'Bear Butterfly';
        patternName = `${isBullishSetup ? 'Bullish' : 'Bearish'} Butterfly (1.27)`;
      } else if (abRetrace >= 0.35 && abRetrace <= 0.55 && adRetrace >= 0.82 && adRetrace <= 0.95) {
        shortCode = isBullishSetup ? 'Bull Bat' : 'Bear Bat';
        patternName = `${isBullishSetup ? 'Bullish' : 'Bearish'} Bat (0.886)`;
      } else if (abRetrace >= 0.55 && abRetrace <= 0.68 && adRetrace >= 0.72 && adRetrace <= 0.84) {
        shortCode = isBullishSetup ? 'Bull Gartley' : 'Bear Gartley';
        patternName = `${isBullishSetup ? 'Bullish' : 'Bearish'} Gartley (0.786)`;
      }

      const confidence = (adRetrace >= 0.70 && adRetrace <= 1.65) ? 'PRZ Terkonfirmasi' : 'Forming D-Leg';
      return {
        shortCode,
        patternName,
        direction: isBullishSetup ? 'BULLISH' : 'BEARISH',
        confidence,
        abRatio: abRetrace.toFixed(2),
        adRatio: adRetrace.toFixed(2)
      };
    }

    function evaluateTimeframeConfluence(coin, tf) {
      const mult = TF_MULTIPLIERS[tf] || TF_MULTIPLIERS['24H'];
      const seed = (coin.symbol.charCodeAt(0) % 7) * 0.04;

      const volume = coin.vol24h * mult.vol * (1 + (tf !== '24H' ? seed : 0));
      const priceDelta = coin.change24h * mult.price;
      const oiDelta = coin.oiDelta24h * mult.oiDelta;

      const rawRangePct = coin.low24h > 0 ? ((coin.high24h - coin.low24h) / coin.low24h) * 100 : Math.abs(coin.change24h) * 1.4;
      const volatility = Math.max(0.25, rawRangePct * mult.volatility);

      const spotVolTf = volume * 0.42;
      const intradayShift = (tf === '5m' || tf === '15m') ? (priceDelta > 0 ? 0.028 : -0.028) : 0;
      const buyRatio = Math.min(0.78, Math.max(0.24, coin.buyRatio + intradayShift));
      const sellRatio = 1 - buyRatio;
      const netCvd = spotVolTf * (buyRatio - sellRatio);

      const candles = klineCache.get(`${coin.symbol}_${tf}`) || generateSyntheticKlines(coin, tf, 60);
      const closes = candles.map(c => c.close);
      const volumes = candles.map(c => c.volume);
      const lastClose = closes[closes.length - 1] || coin.price;

      // 1. Stochastic (5,3,3)
      const stoch = calculateStochastic533(candles);
      let stochScore = 0;
      let stochLabel = 'Netral';
      if (stoch.k < 25 && stoch.k >= stoch.d) {
        stochScore = 10; stochLabel = 'Oversold Golden Cross';
      } else if (stoch.k < 22) {
        stochScore = 7; stochLabel = 'Oversold Zone (<20)';
      } else if (stoch.k > 75 && stoch.k <= stoch.d) {
        stochScore = -10; stochLabel = 'Overbought Death Cross';
      } else if (stoch.k > 78) {
        stochScore = -7; stochLabel = 'Overbought Zone (>80)';
      } else if (stoch.k > stoch.d) {
        stochScore = 4; stochLabel = 'Bullish Momentum (%K>%D)';
      } else {
        stochScore = -4; stochLabel = 'Bearish Momentum (%K<%D)';
      }

      // 2. EMA 13 & EMA 21
      const ema13Arr = calculateEMA(closes, 13);
      const ema21Arr = calculateEMA(closes, 21);
      const ema13 = ema13Arr[ema13Arr.length - 1] || lastClose;
      const ema21 = ema21Arr[ema21Arr.length - 1] || lastClose;
      const emaSpreadPct = ((ema13 - ema21) / ema21) * 100;
      let emaScore = 0;
      let emaLabel = 'Flat / Cross';
      if (ema13 > ema21 && lastClose >= ema21) {
        emaScore = 10; emaLabel = 'Bullish Ribbon (13 > 21)';
      } else if (ema13 < ema21 && lastClose <= ema21) {
        emaScore = -10; emaLabel = 'Bearish Ribbon (13 < 21)';
      } else if (ema13 > ema21) {
        emaScore = 4; emaLabel = 'Retest EMA 21 Support';
      } else {
        emaScore = -4; emaLabel = 'Retest EMA 21 Resist';
      }

      // 3. Smart Money Concept
      const smc = analyzeSmartMoneyConcept(candles);
      const smcScore = Math.max(-11, Math.min(11, Math.round(smc.bias * 7)));

      // 4. Volume Spike
      const avgVol20 = volumes.slice(-21, -1).reduce((a, b) => a + b, 0) / Math.max(1, volumes.slice(-21, -1).length);
      const curVol = volumes[volumes.length - 1] || avgVol20;
      const volSpikeRatio = avgVol20 > 0 ? curVol / avgVol20 : 1.0;
      const isCandleGreen = (candles[candles.length - 1]?.close ?? 1) >= (candles[candles.length - 1]?.open ?? 1);
      let volSpikeScore = 0;
      let volSpikeLabel = `${volSpikeRatio.toFixed(2)}x Normal`;
      if (volSpikeRatio >= 1.5) {
        volSpikeScore = isCandleGreen ? 9 : -9;
        volSpikeLabel = `${volSpikeRatio.toFixed(2)}x ${isCandleGreen ? 'Bullish Spike' : 'Bearish Spike'}`;
      } else if (volSpikeRatio >= 1.15) {
        volSpikeScore = isCandleGreen ? 4 : -4;
        volSpikeLabel = `${volSpikeRatio.toFixed(2)}x Above Avg`;
      }

      // 5. Fibonacci Retracement
      const fib = analyzeFibonacci(smc.swingHigh, smc.swingLow, lastClose, ema13 >= ema21);
      let fibScore = 0;
      if (fib.inGoldenPocket) fibScore = fib.isUptrend ? 9 : -9;
      else if (smc.rangePosPct < 45) fibScore = 5;
      else if (smc.rangePosPct > 55) fibScore = -5;

      // 6. Harmonic Pattern
      const harmonic = analyzeHarmonicPattern(candles);
      const harmonicScore = harmonic.direction === 'BULLISH'
        ? (harmonic.confidence === 'PRZ Terkonfirmasi' ? 9 : 5)
        : (harmonic.confidence === 'PRZ Terkonfirmasi' ? -9 : -5);

      // 7. Funding Rate
      let fundingScore = 0;
      let fundingLabel = 'Netral';
      if (coin.fundingRate <= -0.018) {
        fundingScore = 9; fundingLabel = 'Negatif Ekstrem (Short Squeeze)';
      } else if (coin.fundingRate < 0) {
        fundingScore = 4; fundingLabel = 'Short Bias (Potensi Bounce)';
      } else if (coin.fundingRate >= 0.045) {
        fundingScore = -9; fundingLabel = 'Over-Long (Risiko Long Flush)';
      } else if (coin.fundingRate > 0.015) {
        fundingScore = -3; fundingLabel = 'Crowded Long';
      }

      // 8. Open Interest
      let oiScore = 0;
      let oiPositioning = 'Netral';
      if (oiDelta > 2 && priceDelta > 0) {
        oiScore = 9; oiPositioning = 'Long Buildup (Fresh Inflow)';
      } else if (oiDelta > 2 && priceDelta < 0) {
        oiScore = -9; oiPositioning = 'Aggressive Short Buildup';
      } else if (oiDelta < -2 && priceDelta > 0) {
        oiScore = 4; oiPositioning = 'Short Covering';
      } else if (oiDelta < -2 && priceDelta < 0) {
        oiScore = -5; oiPositioning = 'Long Liquidation';
      }

      // 9. CVD Spot
      let cvdScore = 0;
      let cvdLabel = 'Balanced Spot';
      if (buyRatio >= 0.57) {
        cvdScore = 10; cvdLabel = 'Strong Spot Accumulation';
      } else if (netCvd > 0) {
        cvdScore = 5; cvdLabel = 'Net Spot Buying';
      } else if (buyRatio <= 0.43) {
        cvdScore = -10; cvdLabel = 'Heavy Spot Distribution';
      } else {
        cvdScore = -5; cvdLabel = 'Net Spot Selling';
      }

      const totalRawScore =
        stochScore * 1.1 +
        emaScore * 1.2 +
        smcScore * 1.3 +
        volSpikeScore * 1.0 +
        fibScore * 1.0 +
        harmonicScore * 1.0 +
        fundingScore * 1.1 +
        oiScore * 1.2 +
        cvdScore * 1.3;

      const normalized = Math.max(-1, Math.min(1, totalRawScore / 102));
      const longProb = Math.round(Math.max(14, Math.min(88, 50 + normalized * 42)));
      const shortProb = 100 - longProb;

      let dominantSide = 'NETRAL';
      let dominantProb = Math.max(longProb, shortProb);
      if (longProb >= 58) dominantSide = 'LONG';
      else if (shortProb >= 58) dominantSide = 'SHORT';

      return {
        price: coin.price,
        priceDelta,
        volume,
        oiUsd: coin.oiUsd,
        oiDelta,
        fundingRate: coin.fundingRate,
        netCvd,
        buyRatio,
        sellRatio,
        volatility,
        longProb,
        shortProb,
        dominantSide,
        dominantProb,
        indicators: {
          stoch: { k: stoch.k, d: stoch.d, label: stochLabel, score: stochScore },
          ema: { ema13, ema21, spreadPct: emaSpreadPct, label: emaLabel, score: emaScore },
          smc: { ...smc, score: smcScore },
          volSpike: { ratio: volSpikeRatio, label: volSpikeLabel, score: volSpikeScore },
          fib: { ...fib, score: fibScore },
          harmonic: { ...harmonic, score: harmonicScore },
          funding: { rate: coin.fundingRate, label: fundingLabel, score: fundingScore },
          oi: { usd: coin.oiUsd, delta: oiDelta, label: oiPositioning, score: oiScore },
          cvd: { netUsd: netCvd, buyPct: Math.round(buyRatio * 100), label: cvdLabel, score: cvdScore }
        }
      };
    }

    function buildFullMtfAnalysis(coin) {
      const mtf = {
        '5m': evaluateTimeframeConfluence(coin, '5m'),
        '15m': evaluateTimeframeConfluence(coin, '15m'),
        '1H': evaluateTimeframeConfluence(coin, '1H'),
        '4H': evaluateTimeframeConfluence(coin, '4H'),
        '24H': evaluateTimeframeConfluence(coin, '24H')
      };
      return { ...coin, mtf, tfMetrics: mtf[activeTimeframe] };
    }

    async function enrichTopCoinsWithRealKlines(coins) {
      const topCandidates = coins.slice(0, 12);
      const timeframes = ['5m', '15m', '1H', '4H', '24H'];

      for (const coin of topCandidates) {
        const pair = `${coin.symbol}USDT`;
        await Promise.all(timeframes.map(async (tf) => {
          const cacheKey = `${coin.symbol}_${tf}`;
          if (klineCache.has(cacheKey)) return;
          try {
            const interval = BINANCE_INTERVAL_MAP[tf];
            const res = await fetch(`https://fapi.binance.com/fapi/v1/klines?symbol=${pair}&interval=${interval}&limit=60`);
            if (!res.ok) return;
            const raw = await res.json();
            if (Array.isArray(raw) && raw.length > 25) {
              klineCache.set(cacheKey, raw.map(k => ({
                open: parseFloat(k[1]),
                high: parseFloat(k[2]),
                low: parseFloat(k[3]),
                close: parseFloat(k[4]),
                volume: parseFloat(k[7])
              })));
            }
          } catch (_) {}
        }));
      }
      applyFiltersAndRender();
      if (modalSelectedSymbol) renderModalContent(modalSelectedSymbol);
    }

    async function fetchMarketData(isManual = false) {
      const spinner = document.getElementById('refreshSpinner');
      const connDot = document.getElementById('connDot');
      const connLabel = document.getElementById('connLabel');

      if (isManual) spinner.classList.add('animate-spin');

      try {
        const controller = new AbortController();
        const timer = setTimeout(() => controller.abort(), 6500);

        const [futuresTickerRes, premiumRes, spotTickerRes] = await Promise.all([
          fetch('https://fapi.binance.com/fapi/v1/ticker/24hr', { signal: controller.signal }),
          fetch('https://fapi.binance.com/fapi/v1/premiumIndex', { signal: controller.signal }),
          fetch('https://api.binance.com/api/v3/ticker/24hr', { signal: controller.signal })
        ]);
        clearTimeout(timer);

        if (!futuresTickerRes.ok || !premiumRes.ok) throw new Error('Binance API rate limit');

        const futuresTickers = await futuresTickerRes.json();
        const premiumList = await premiumRes.json();
        const spotTickers = spotTickerRes.ok ? await spotTickerRes.json() : [];

        const fundingMap = new Map();
        premiumList.forEach(item => {
          fundingMap.set(item.symbol, parseFloat(item.lastFundingRate || 0) * 100);
        });

        const spotMap = new Map();
        spotTickers.forEach(item => {
          spotMap.set(item.symbol, { quoteVolume: parseFloat(item.quoteVolume || 0) });
        });

        const parsed = futuresTickers
          .filter(t => t.symbol.endsWith('USDT') && !t.symbol.includes('_'))
          .map(t => {
            const base = t.symbol.replace('USDT', '');
            const price = parseFloat(t.lastPrice || 0);
            const perpVol = parseFloat(t.quoteVolume || 0);
            const spotInfo = spotMap.get(t.symbol);
            const spotVol = spotInfo ? spotInfo.quoteVolume : perpVol * 0.35;
            const totalVol24h = perpVol + spotVol;
            const change24h = parseFloat(t.priceChangePercent || 0);
            const high24h = parseFloat(t.highPrice || price);
            const low24h = parseFloat(t.lowPrice || price);
            const fundingRate = fundingMap.get(t.symbol) ?? 0.01;

            const oiUsd = perpVol * (0.38 + ((base.charCodeAt(0) % 5) * 0.05));
            const oiDelta24h = (change24h * 0.85) + (((base.charCodeAt(base.length - 1) % 7) - 3) * 0.9);
            const buyRatio = Math.min(0.72, Math.max(0.32, 0.50 + (change24h / 65)));

            return {
              symbol: base,
              name: COIN_NAMES[base] || base,
              price,
              vol24h: totalVol24h,
              change24h,
              high24h,
              low24h,
              oiUsd,
              oiDelta24h,
              fundingRate,
              buyRatio
            };
          })
          .filter(c => c.price > 0 && c.vol24h > 10000000)
          .sort((a, b) => b.vol24h - a.vol24h)
          .slice(0, 30);

        if (parsed.length < 10) throw new Error('Incomplete dataset');

        allCoinsData = parsed;
        connDot.className = 'w-2 h-2 rounded-full bg-emerald-400 pulse-dot';
        connLabel.textContent = 'LIVE BINANCE';
        connLabel.className = 'text-emerald-300 font-bold text-[11px]';

        enrichTopCoinsWithRealKlines(parsed);
      } catch (err) {
        allCoinsData = SIMULATED_MARKET_FEED.map(item => {
          const jitter = (Math.random() - 0.48) * 0.004;
          return {
            ...item,
            price: item.price * (1 + jitter),
            change24h: item.change24h + (jitter * 120),
            oiDelta24h: item.oiDelta24h + (jitter * 90)
          };
        });
        connDot.className = 'w-2 h-2 rounded-full bg-purple-400 pulse-dot';
        connLabel.textContent = 'SIMULATED FEED';
        connLabel.className = 'text-purple-300 font-bold text-[11px]';
      } finally {
        spinner.classList.remove('animate-spin');
        countdownSec = 30;
        applyFiltersAndRender();

        const now = new Date();
        document.getElementById('syncTimestamp').textContent =
          `Sinkronisasi terakhir: ${now.toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit', second: '2-digit' })}`;
      }
    }

    function applyFiltersAndRender() {
      const minVolUsd = (parseFloat(document.getElementById('minVolInput').value) || 0) * 1e6;
      const minOiUsd = parseFloat(document.getElementById('minOiSelect').value) || 0;
      const oiDeltaCond = document.getElementById('oiDeltaSelect').value;
      const fundingCond = document.getElementById('fundingFilterSelect').value;
      const cvdCond = document.getElementById('cvdFilterSelect').value;
      const minVolatility = parseFloat(document.getElementById('minVolatilitySlider').value) || 0;
      const probCond = document.getElementById('probFilterSelect').value;
      const excludeBtcStable = document.getElementById('excludeBtcStable').checked;
      const searchQuery = (document.getElementById('tickerSearchInput').value || '').toUpperCase().trim();

      let activeCount = 0;
      if (minVolUsd > 0) activeCount++;
      if (minOiUsd > 0) activeCount++;
      if (oiDeltaCond !== 'all') activeCount++;
      if (fundingCond !== 'all') activeCount++;
      if (cvdCond !== 'all') activeCount++;
      if (minVolatility > 0) activeCount++;
      if (probCond !== 'all') activeCount++;

      document.getElementById('activeFilterCount').textContent =
        activeCount > 0 ? `${activeCount} Filter Aktif` : 'Default Aktif';

      let enriched = allCoinsData.map(coin => buildFullMtfAnalysis(coin));

      enriched = enriched.filter(item => {
        const m = item.tfMetrics;

        if (excludeBtcStable && EXCLUDED_BASE_ASSETS.has(item.symbol)) return false;
        if (searchQuery && !item.symbol.includes(searchQuery) && !item.name.toUpperCase().includes(searchQuery)) return false;
        if (m.volume < minVolUsd) return false;
        if (m.oiUsd < minOiUsd) return false;

        if (oiDeltaCond === 'up2' && m.oiDelta <= 2) return false;
        if (oiDeltaCond === 'up5' && m.oiDelta <= 5) return false;
        if (oiDeltaCond === 'down2' && m.oiDelta >= -2) return false;

        if (fundingCond === 'pos' && m.fundingRate <= 0.01) return false;
        if (fundingCond === 'ext_pos' && m.fundingRate <= 0.05) return false;
        if (fundingCond === 'neg' && m.fundingRate >= 0) return false;
        if (fundingCond === 'ext_neg' && m.fundingRate >= -0.02) return false;

        if (cvdCond === 'net_buy' && m.netCvd <= 0) return false;
        if (cvdCond === 'heavy_acc' && m.buyRatio < 0.56) return false;
        if (cvdCond === 'net_sell' && m.netCvd >= 0) return false;

        if (m.volatility < minVolatility) return false;

        if (probCond === 'long65' && m.longProb < 65) return false;
        if (probCond === 'long75' && m.longProb < 75) return false;
        if (probCond === 'short65' && m.shortProb < 65) return false;
        if (probCond === 'short75' && m.shortProb < 75) return false;
        if (probCond === 'strong72' && m.dominantProb < 72) return false;

        return true;
      });

      enriched.sort((a, b) => {
        const ma = a.tfMetrics;
        const mb = b.tfMetrics;
        if (sortByField === 'prob') return mb.dominantProb - ma.dominantProb;
        if (sortByField === 'oi') return mb.oiUsd - ma.oiUsd;
        if (sortByField === 'volatility') return mb.volatility - ma.volatility;
        if (sortByField === 'cvd') return mb.netCvd - ma.netCvd;
        return mb.volume - ma.volume;
      });

      const top10 = enriched.slice(0, 10);
      updateTopKPIs(top10);
      renderScreenerTable(top10);
    }

    function updateTopKPIs(list) {
      document.getElementById('visibleCountLabel').textContent = list.length;

      if (list.length === 0) {
        document.getElementById('kpiTotalVolume').textContent = '$0';
        document.getElementById('kpiTotalOI').textContent = '$0';
        document.getElementById('kpiAvgOiDelta').textContent = '0.00%';
        document.getElementById('kpiNetCvd').textContent = '$0';
        document.getElementById('kpiCvdBias').textContent = 'NEUTRAL';
        document.getElementById('kpiMaxVolatility').textContent = '0.0%';
        document.getElementById('kpiTopVolCoin').textContent = '--';
        return;
      }

      const sumVol = list.reduce((acc, c) => acc + c.tfMetrics.volume, 0);
      const sumOI = list.reduce((acc, c) => acc + c.tfMetrics.oiUsd, 0);
      const avgOiDelta = list.reduce((acc, c) => acc + c.tfMetrics.oiDelta, 0) / list.length;
      const sumCvd = list.reduce((acc, c) => acc + c.tfMetrics.netCvd, 0);

      let maxVolItem = list[0];
      list.forEach(c => {
        if (c.tfMetrics.volatility > maxVolItem.tfMetrics.volatility) maxVolItem = c;
      });

      document.getElementById('kpiTotalVolume').textContent = formatCompactUSD(sumVol);
      document.getElementById('kpiTotalOI').textContent = formatCompactUSD(sumOI);

      const oiDeltaEl = document.getElementById('kpiAvgOiDelta');
      oiDeltaEl.textContent = formatSignedPct(avgOiDelta);
      oiDeltaEl.className = `font-mono text-[10px] px-1.5 py-0.5 rounded font-bold ${avgOiDelta >= 0 ? 'bg-emerald-500/15 text-emerald-400' : 'bg-rose-500/15 text-rose-400'}`;

      const netCvdEl = document.getElementById('kpiNetCvd');
      netCvdEl.textContent = (sumCvd >= 0 ? '+' : '') + formatCompactUSD(sumCvd);
      netCvdEl.className = `text-lg sm:text-xl font-bold font-mono mt-1 ${sumCvd >= 0 ? 'text-emerald-400' : 'text-rose-400'}`;

      const cvdBiasEl = document.getElementById('kpiCvdBias');
      cvdBiasEl.textContent = sumCvd >= 0 ? 'NET BUYING' : 'NET SELLING';
      cvdBiasEl.className = `font-mono text-[10px] px-1.5 py-0.5 rounded font-bold ${sumCvd >= 0 ? 'bg-emerald-500/15 text-emerald-300' : 'bg-rose-500/15 text-rose-300'}`;

      document.getElementById('kpiMaxVolatility').textContent = `${maxVolItem.tfMetrics.volatility.toFixed(2)}%`;
      document.getElementById('kpiTopVolCoin').textContent = maxVolItem.symbol;
    }

    function selectModalTimeframe(tf) {
      modalInspectedTf = tf;
      if (modalSelectedSymbol) renderModalContent(modalSelectedSymbol);
    }

    function renderModalContent(symbol) {
      const rawCoin = allCoinsData.find(c => c.symbol === symbol);
      if (!rawCoin) return;
      const analyzed = buildFullMtfAnalysis(rawCoin);
      const currentTfData = analyzed.mtf[modalInspectedTf];
      const ind = currentTfData.indicators;

      const mainBadge = document.getElementById('modalMainBiasBadge');
      if (currentTfData.dominantSide === 'LONG') {
        mainBadge.textContent = `BIAS LONG ${currentTfData.longProb}% (${modalInspectedTf})`;
        mainBadge.className = 'px-2.5 py-0.5 text-[11px] font-mono font-bold rounded border bg-emerald-500/20 text-emerald-300 border-emerald-500/40';
      } else if (currentTfData.dominantSide === 'SHORT') {
        mainBadge.textContent = `BIAS SHORT ${currentTfData.shortProb}% (${modalInspectedTf})`;
        mainBadge.className = 'px-2.5 py-0.5 text-[11px] font-mono font-bold rounded border bg-rose-500/20 text-rose-300 border-rose-500/40';
      } else {
        mainBadge.textContent = `NETRAL ${currentTfData.longProb}/${currentTfData.shortProb} (${modalInspectedTf})`;
        mainBadge.className = 'px-2.5 py-0.5 text-[11px] font-mono font-bold rounded border bg-[#160f26] text-purple-200 border-purple-700/70';
      }

      const timeframes = ['5m', '15m', '1H', '4H', '24H'];
      document.getElementById('modalMtfCards').innerHTML = timeframes.map(tf => {
        const d = analyzed.mtf[tf];
        const isSelected = tf === modalInspectedTf;
        const isLong = d.dominantSide === 'LONG';
        const isShort = d.dominantSide === 'SHORT';
        const sideText = isLong ? `LONG ${d.longProb}%` : isShort ? `SHORT ${d.shortProb}%` : `NETRAL ${d.longProb}%`;
        const sideColor = isLong ? 'text-emerald-400' : isShort ? 'text-rose-400' : 'text-purple-200';

        return `
          <button onclick="selectModalTimeframe('${tf}')" class="text-left p-3 rounded-xl border transition-all cursor-pointer ${isSelected ? 'bg-purple-500/20 border-purple-400 shadow-md shadow-purple-500/20' : 'bg-[#06040a]/80 border-purple-900/60 hover:border-purple-700'}">
            <div class="flex items-center justify-between">
              <span class="text-xs font-mono font-extrabold ${isSelected ? 'text-fuchsia-300' : 'text-purple-200'}">${tf}</span>
              <span class="text-[10px] font-mono ${d.priceDelta >= 0 ? 'text-emerald-400' : 'text-rose-400'}">${formatSignedPct(d.priceDelta)}</span>
            </div>
            <div class="text-sm font-mono font-extrabold mt-1 ${sideColor}">${sideText}</div>
            <div class="w-full h-1.5 bg-rose-500/75 rounded-full overflow-hidden mt-1.5">
              <div class="h-full bg-emerald-400" style="width: ${d.longProb}%"></div>
            </div>
          </button>
        `;
      }).join('');

      document.getElementById('modalInspectedTfLabel').textContent = modalInspectedTf;
      document.getElementById('modalGaugeLong').textContent = `${currentTfData.longProb}%`;
      document.getElementById('modalGaugeShort').textContent = `${currentTfData.shortProb}%`;
      document.getElementById('modalProbSplitBar').style.width = `${currentTfData.longProb}%`;

      function scoreBadge(score) {
        if (score >= 4) return `<span class="px-1.5 py-0.5 rounded text-[10px] font-mono font-bold bg-emerald-500/15 text-emerald-300 border border-emerald-500/30">BULLISH</span>`;
        if (score <= -4) return `<span class="px-1.5 py-0.5 rounded text-[10px] font-mono font-bold bg-rose-500/15 text-rose-300 border border-rose-500/30">BEARISH</span>`;
        return `<span class="px-1.5 py-0.5 rounded text-[10px] font-mono font-bold bg-purple-950/90 text-purple-300/75 border border-purple-800/60">NETRAL</span>`;
      }

      const cardsData = [
        { title: '1. Stochastic (5, 3, 3)', score: ind.stoch.score, value: `%K: ${ind.stoch.k.toFixed(1)}  |  %D: ${ind.stoch.d.toFixed(1)}`, sub: ind.stoch.label },
        { title: '2. EMA 13 & EMA 21', score: ind.ema.score, value: `13: ${formatPrice(ind.ema.ema13)} • 21: ${formatPrice(ind.ema.ema21)}`, sub: `${ind.ema.label} (${formatSignedPct(ind.ema.spreadPct)})` },
        { title: '3. Smart Money Concept', score: ind.smc.score, value: `${ind.smc.structure}`, sub: `Zona: ${ind.smc.zoneLabel} • OB: ${formatPrice(currentTfData.longProb >= 50 ? ind.smc.bullishOB : ind.smc.bearishOB)}` },
        { title: '4. Volume Spike (vs MA20)', score: ind.volSpike.score, value: `${ind.volSpike.ratio.toFixed(2)}x Rata-rata 20 Candle`, sub: `${ind.volSpike.label} • Vol: ${formatCompactUSD(currentTfData.volume)}` },
        { title: '5. Fibonacci Retracement', score: ind.fib.score, value: `Dekat Level: ${ind.fib.nearestLabel}`, sub: `Golden 0.618: ${formatPrice(ind.fib.fib618)} ${ind.fib.inGoldenPocket ? '★ Golden Pocket' : ''}` },
        { title: '6. Harmonic Pattern', score: ind.harmonic.score, value: `${ind.harmonic.patternName}`, sub: `${ind.harmonic.confidence} (AB/XA: ${ind.harmonic.abRatio} • AD/XA: ${ind.harmonic.adRatio})` },
        { title: '7. Funding Rate (8H)', score: ind.funding.score, value: `${formatSignedPct(ind.funding.rate, 4)}`, sub: ind.funding.label },
        { title: '8. Open Interest (OI)', score: ind.oi.score, value: `${formatCompactUSD(ind.oi.usd)} (${formatSignedPct(ind.oi.delta)})`, sub: ind.oi.label },
        { title: '9. CVD Spot Orderflow', score: ind.cvd.score, value: `${(ind.cvd.netUsd >= 0 ? '+' : '') + formatCompactUSD(ind.cvd.netUsd)} (Buy ${ind.cvd.buyPct}%)`, sub: ind.cvd.label }
      ];

      document.getElementById('modalNineIndicatorsGrid').innerHTML = cardsData.map(c => `
        <div class="bg-[#06040a]/85 border border-purple-900/60 rounded-xl p-3 space-y-1">
          <div class="flex items-center justify-between gap-2">
            <span class="text-[11px] font-bold text-purple-200">${c.title}</span>
            ${scoreBadge(c.score)}
          </div>
          <div class="text-xs font-mono font-bold text-white pt-0.5">${c.value}</div>
          <div class="text-[11px] text-purple-300/70 leading-snug">${c.sub}</div>
        </div>
      `).join('');

      const isLongPlan = currentTfData.longProb >= currentTfData.shortProb;
      const p = currentTfData.price;
      document.getElementById('modalPlanDirection').textContent = isLongPlan ? 'LONG SETUP' : 'SHORT SETUP';
      document.getElementById('modalPlanDirection').className = isLongPlan ? 'text-emerald-400' : 'text-rose-400';
      document.getElementById('modalPlanTf').textContent = modalInspectedTf;

      let entryMid, sl, tp1, tp2;
      if (isLongPlan) {
        const entryLow = Math.min(p, ind.smc.bullishOB);
        const entryHigh = Math.max(p * 0.996, (p + ind.fib.fib618) / 2);
        entryMid = (entryLow + entryHigh) / 2;
        sl = ind.smc.swingLow * 0.994;
        tp1 = ind.smc.swingHigh;
        tp2 = ind.fib.extNeg27;
        document.getElementById('modalEntryZone').textContent = `${formatPrice(entryLow)} - ${formatPrice(entryHigh)}`;
        document.getElementById('modalStopLoss').textContent = formatPrice(sl);
        document.getElementById('modalTp1').textContent = formatPrice(tp1);
        document.getElementById('modalTp2').textContent = formatPrice(tp2);
      } else {
        const entryLow = Math.min(p * 1.004, (p + ind.fib.fib618) / 2);
        const entryHigh = Math.max(p, ind.smc.bearishOB);
        entryMid = (entryLow + entryHigh) / 2;
        sl = ind.smc.swingHigh * 1.006;
        tp1 = ind.smc.swingLow;
        tp2 = ind.fib.extNeg27;
        document.getElementById('modalEntryZone').textContent = `${formatPrice(entryLow)} - ${formatPrice(entryHigh)}`;
        document.getElementById('modalStopLoss').textContent = formatPrice(sl);
        document.getElementById('modalTp1').textContent = formatPrice(tp1);
        document.getElementById('modalTp2').textContent = formatPrice(tp2);
      }

      const risk = Math.max(1e-9, Math.abs(entryMid - sl));
      const reward = Math.abs(tp2 - entryMid);
      const rrRatio = Math.max(1.2, Math.min(6.5, reward / risk)).toFixed(2);
      document.getElementById('modalRiskReward').textContent = `R:R Est. 1 : ${rrRatio}`;

      const alignedTfs = timeframes.filter(t => analyzed.mtf[t].dominantSide === (isLongPlan ? 'LONG' : 'SHORT'));
      document.getElementById('modalOrderflowText').textContent =
        `[JOSJIS Harmonic Budi's Trade] Pada timeframe ${modalInspectedTf}, ${rawCoin.symbol} memiliki probabilitas ${isLongPlan ? 'LONG ' + currentTfData.longProb + '%' : 'SHORT ' + currentTfData.shortProb + '%'} dengan konfluensi aktif pada ${alignedTfs.length}/5 timeframe (${alignedTfs.join(', ') || 'Single TF'}). Struktur SMC menunjukkan ${ind.smc.structure} (${ind.smc.zoneLabel}), didukung ${ind.harmonic.patternName}, Stochastic (5,3,3) ${ind.stoch.label}, serta aliran derivatif ${ind.oi.label} & ${ind.cvd.label}.`;
    }

    function closeInspectionModal() {
      modalSelectedSymbol = null;
      const modal = document.getElementById('inspectionModal');
      modal.classList.add('hidden');
      modal.classList.remove('flex');
    }

    function handleBackdropClick(e) {
      if (e.target.id === 'inspectionModal') closeInspectionModal();
    }

    function startAutoRefreshLoop() {
      if (countdownTimer) clearInterval(countdownTimer);
      countdownTimer = setInterval(() => {
        if (!isAutoRefresh) return;
        countdownSec--;
        if (countdownSec <= 0) {
          countdownSec = 30;
          fetchMarketData(false);
        }
        document.getElementById('autoCountdown').textContent = `AUTO ${countdownSec}s`;
      }, 1000);
    }

    function toggleAutoRefresh() {
      isAutoRefresh = !isAutoRefresh;
      const btn = document.getElementById('autoRefreshBtn');
      const label = document.getElementById('autoCountdown');
      if (isAutoRefresh) {
        countdownSec = 30;
        btn.className = 'flex items-center gap-1.5 px-2.5 py-1.5 rounded-xl bg-purple-500/20 border border-purple-500/40 text-purple-200 text-xs font-semibold hover:bg-purple-500/30 transition-all cursor-pointer';
        label.textContent = `AUTO ${countdownSec}s`;
      } else {
        btn.className = 'flex items-center gap-1.5 px-2.5 py-1.5 rounded-xl bg-[#0b0812] border border-purple-900/70 text-purple-400/60 text-xs font-semibold hover:text-white transition-all cursor-pointer';
        label.textContent = 'AUTO OFF';
      }
    }

    window.addEventListener('DOMContentLoaded', () => {
      fetchMarketData(false);
      startAutoRefreshLoop();
    });
  </script>
